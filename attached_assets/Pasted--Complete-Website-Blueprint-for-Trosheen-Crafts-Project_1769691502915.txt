<script>
	import { onMount, onDestroy } from 'svelte';
	
	export let speed = 0.5; // Parallax speed multiplier
	export let direction = 'vertical'; // 'vertical' | 'horizontal'
	
	let container;
	let translateY = 0;
	let translateX = 0;
	
	function handleScroll() {
		if (!container) return;
		
		const rect = container.getBoundingClientRect();
		const scrollProgress = -rect.top / (rect.height + window.innerHeight);
		
		if (direction === 'vertical') {
			translateY = scrollProgress * 100 * speed;
		} else {
			translateX = scrollProgress * 100 * speed;
		}
	}
	
	onMount(() => {
		window.addEventListener('scroll', handleScroll, { passive: true });
		handleScroll();
	});
	
	onDestroy(() => {
		window.removeEventListener('scroll', handleScroll);
	});
</script>

<div 
	bind:this={container}
	class="parallax-container"
	style="transform: translate3d({translateX}px, {translateY}px, 0);"

	<slot />

<script>
	import { onMount } from 'svelte';
	import { fade, fly } from 'svelte/transition';
	
	export let delay = 0;
	export let duration = 800;
	export let threshold = 0.2;
	export let animation = 'fade-up'; // 'fade-up' | 'fade-in' | 'scale-in'
	
	let isVisible = false;
	let element;
	
	onMount(() => {
		const observer = new IntersectionObserver(
			([entry]) => {
				if (entry.isIntersecting) {
					setTimeout(() => {
						isVisible = true;
					}, delay);
					observer.disconnect();
				}
			},
			{ threshold }
		);
		
		if (element) {
			observer.observe(element);
		}
		
		return () => observer.disconnect();
	});
	
	function getTransition() {
		switch(animation) {
			case 'fade-up':
				return { y: 60, duration, delay: 0 };
			case 'scale-in':
				return { scale: 0.8, duration, delay: 0 };
			default:
				return { duration, delay: 0 };
		}
	}
</script>

<div bind:this={element}>
	{#if isVisible}
		<div in:fly={getTransition()}>
			<slot />
		</div>
	{:else}
		<div style="opacity: 0;">
			<slot />
		</div>
	{/if}
</div>

import { writable } from 'svelte/store';
import { nanoid } from 'nanoid';

function createCart() {
	const { subscribe, set, update } = writable({
		items: [],
		total: 0
	});
	
	return {
		subscribe,
		add: (product, quantity = 1) => {
			update(cart => {
				const existingItem = cart.items.find(item => item.id === product.id);
				
				if (existingItem) {
					existingItem.quantity += quantity;
				} else {
					cart.items.push({
						...product,
						cartId: nanoid(),
						quantity
					});
				}
				
				cart.total = cart.items.reduce((sum, item) => sum + (item.price * item.quantity), 0);
				return cart;
			});
		},
		remove: (cartId) => {
			update(cart => {
				cart.items = cart.items.filter(item => item.cartId !== cartId);
				cart.total = cart.items.reduce((sum, item) => sum + (item.price * item.quantity), 0);
				return cart;
			});
		},
		updateQuantity: (cartId, quantity) => {
			update(cart => {
				const item = cart.items.find(item => item.cartId === cartId);
				if (item) {
					item.quantity = Math.max(0, quantity);
					if (item.quantity === 0) {
						cart.items = cart.items.filter(i => i.cartId !== cartId);
					}
				}
				cart.total = cart.items.reduce((sum, item) => sum + (item.price * item.quantity), 0);
				return cart;
			});
		},
		clear: () => set({ items: [], total: 0 })
	};
}

export const cart = createCart();

import gsap from 'gsap';
import ScrollTrigger from 'gsap/ScrollTrigger';

export function initScrollAnimations() {
	gsap.registerPlugin(ScrollTrigger);
	
	// Generic scroll reveal
	gsap.utils.toArray('.scroll-reveal').forEach(element => {
		gsap.from(element, {
			scrollTrigger: {
				trigger: element,
				start: 'top 80%',
				end: 'top 20%',
				toggleActions: 'play none none reverse'
			},
			y: 60,
			opacity: 0,
			duration: 1,
			ease: 'power3.out'
		});
	});
}

export function parallaxScroll(element, speed = 0.5) {
	gsap.to(element, {
		scrollTrigger: {
			trigger: element,
			scrub: true
		},
		y: (i, target) => -ScrollTrigger.maxScroll(window) * speed,
		ease: 'none'
	});
}

export function fadeInStagger(selector, staggerAmount = 0.15) {
	return gsap.from(selector, {
		opacity: 0,
		y: 50,
		stagger: staggerAmount,
		duration: 0.8,
		ease: 'power2.out'
	});
}
